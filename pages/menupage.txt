    <script>
        function incrementQuantity(visibleId, hiddenId, maxLimit = 10) {
            const quantityElem = document.getElementById(visibleId);
            const hiddenInput = document.getElementById(hiddenId);
            let quantity = parseInt(quantityElem.textContent);

            if (quantity < maxLimit) {
                quantity += 1;
                quantityElem.textContent = quantity;
                hiddenInput.value = quantity;
            } else {
                alert(`Maximum quantity limit of ${maxLimit} reached`);
            }
        }

        function decrementQuantity(visibleId, hiddenId, minLimit = 1) {
            const quantityElem = document.getElementById(visibleId);
            const hiddenInput = document.getElementById(hiddenId);
            let quantity = parseInt(quantityElem.textContent);

            if (quantity > minLimit) {
                quantity -= 1;
                quantityElem.textContent = quantity;
                hiddenInput.value = quantity;
            } else {
                alert(`Minimum quantity limit of ${minLimit} reached`);
            }
        }
    </script>
